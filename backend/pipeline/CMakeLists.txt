cmake_minimum_required(VERSION 3.20)
project(douyin_pipeline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(WIN32)
  include(windows.cmake)
  set_source_files_properties(portable.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/portable.manifest")
else()
  find_package(Protobuf REQUIRED)
  find_package(Boost REQUIRED COMPONENTS system)
  find_package(OpenSSL REQUIRED)
endif()
find_package(Threads REQUIRED)
if(WIN32)
  foreach(proto douyin ingest)
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${proto}.pb.cc ${CMAKE_CURRENT_BINARY_DIR}/${proto}.pb.h
      COMMAND $<TARGET_FILE:protobuf::protoc> --cpp_out=${CMAKE_CURRENT_BINARY_DIR} -I${CMAKE_CURRENT_SOURCE_DIR}/../proto ${CMAKE_CURRENT_SOURCE_DIR}/../proto/${proto}.proto
      DEPENDS protobuf::protoc ../proto/${proto}.proto VERBATIM)
    list(APPEND PROTO_SRCS ${CMAKE_CURRENT_BINARY_DIR}/${proto}.pb.cc)
  endforeach()
  add_library(pipeline_protocol STATIC ${PROTO_SRCS})
else()
  protobuf_generate_cpp(PROTO_SRCS PROTO_HDRS ../proto/douyin.proto ../proto/ingest.proto)
  add_library(pipeline_protocol STATIC ${PROTO_SRCS})
endif()
target_include_directories(pipeline_protocol PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(pipeline_protocol PUBLIC protobuf::libprotobuf)
function(pipeline_target name source)
  add_executable(${name} ${source})
  if(WIN32)
    target_sources(${name} PRIVATE portable.rc)
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(${name} PRIVATE pipeline_protocol Threads::Threads portable_dependencies)
  else()
    target_link_libraries(${name} PRIVATE pipeline_protocol Boost::system Threads::Threads OpenSSL::Crypto cpp-httplib sqlite3 rabbitmq hiredis z)
  endif()
endfunction()
pipeline_target(douyin-pipeline main.cpp)
if(WIN32)
  add_executable(portable-host ../../portable/host.cpp portable.rc)
  target_include_directories(portable-host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  if(MINGW)
    target_link_options(portable-host PRIVATE -municode)
  endif()
  set_target_properties(portable-host PROPERTIES OUTPUT_NAME DouyinDanmu)
endif()
